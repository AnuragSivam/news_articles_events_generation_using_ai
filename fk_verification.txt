SELECT
    hash_id,
    COUNT(*)
FROM mzamin_articles_nlp
GROUP BY hash_id
HAVING COUNT(*) > 1;
ALTER TABLE mzamin_articles_nlp
ADD CONSTRAINT uq_mzamin_hash_id
UNIQUE(hash_id);
ALTER TABLE hash_id_foreign_key
ADD CONSTRAINT fk_hash_id_foreign_key
FOREIGN KEY (hash_id)
REFERENCES mzamin_articles_nlp(hash_id);
SELECT
    conname,
    contype
FROM pg_constraint
WHERE conrelid = 'hash_id_foreign_key'::regclass;
SELECT
    tc.constraint_name,
    kcu.column_name,
    ccu.table_name AS referenced_table,
    ccu.column_name AS referenced_column
FROM information_schema.table_constraints tc
JOIN information_schema.key_column_usage kcu
    ON tc.constraint_name = kcu.constraint_name
JOIN information_schema.constraint_column_usage ccu
    ON ccu.constraint_name = tc.constraint_name
WHERE tc.table_name = 'hash_id_foreign_key'
AND tc.constraint_type = 'FOREIGN KEY';
SELECT
    h.hash_id,
    h.event_id,
    m.headline_english
FROM hash_id_foreign_key h
JOIN mzamin_articles_nlp m
    ON h.hash_id = m.hash_id
ORDER BY h.hash_id ASC
LIMIT 20;
